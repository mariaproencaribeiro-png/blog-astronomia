// Função para alternar entre Modo Claro e Escuro function toggleTheme() { const body = document.body; const themeIcon = document.getElementById('theme-icon'); const themeText = document.getElementById('theme-text'); body.classList.toggle('light-mode'); if (body.classList.contains('light-mode')) { themeIcon.className = "fa-solid fa-sun"; themeText.textContent = "Modo Claro"; } else { themeIcon.className = "fa-solid fa-moon"; themeText.textContent = "Modo Escuro"; } }
